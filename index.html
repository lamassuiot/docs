# 1. Find your primary interface IP
$ ip a

# 2. Deploy Lamassu with that IP
$ curl -fsSL "https://raw.githubusercontent.com/lamassuiot/lamassu-helm/refs/heads/main/scripts/lamassu-fast-lane.sh" | bash -s -- \
-n \
--domain lamassu.mypki \
--sample-data \
-ip 192.168.1.101
# → cluster ready
